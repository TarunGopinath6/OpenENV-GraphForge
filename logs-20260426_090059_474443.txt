class FenwickTree2D:
    """
    2D Fenwick Tree (Binary Indexed Tree) implementation.

    Supports:
    - add(r, c, val): Add value at position (r, c) in O(log R * log C)
    - query(r1, c1, r2, c2): Query sum in region [r1, r2) x [c1, c2) in O(log R * log C)

    More info: https://en.wikipedia.org/wiki/Binary_indexed_tree
    """

    def __init__(self, rows: int, cols: int) -> None:
        """
        Constructor for the 2D Fenwick tree

        Parameters:
            rows (int): number of rows
            cols (int): number of columns

        """
        self.rows = rows
        self.cols = cols
        # Use 1-based indexing internally
        # tree[i][j] stores sum for node at 1-based coordinate (i, j)
        self.tree: list[list[int]] = [[0] * (cols + 1) for _ in range(rows + 1)]

    def add(self, r: int, c: int, val: int) -> None:
        """
        Add a value to position (r, c) in O(log R * log C)

        Parameters:
            r (int): row index (0-based)
            c (int): column index (0-based)
            val (int): value to add

        Returns:
            None

        """
        r += 1  # Convert to 1-based index
        c += 1  # Convert to 1-based index

        i = r
        while i <= self.rows:
            j = c
            while j <= self.cols:
                self.tree[i][j] += val
                j += j & -j
            i += i & -i

    def query(self, r1: int, c1: int, r2: int, c2: int) -> int:
        """
        Query the sum of all elements in region [r1, r2) x [c1, c2)
        in O(log R * log C)

        Parameters:
            r1 (int): left/bottom row bound (inclusive)
            c1 (int): left/top column bound (inclusive)
            r2 (int): right/bottom row bound (exclusive)
            c2 (int): right/top column bound (exclusive)

        Returns:
            int: sum of all elements in the specified region

        """
        # Inclusion-exclusion for 2D range query
        return (self._prefix_sum(r2 - 1, c2 - 1) - 
                self._prefix_sum(r1 - 1, c2 - 1) - 
                self._prefix_sum(r2 - 1, c1 - 1) + 
                self._prefix_sum(r1 - 1, c1 - 1))

    def _prefix_sum(self, r: int, c: int) -> int:
        """
        Compute prefix sum from (0, 0) to (r, c) in O(log R * log C)

        Parameters:
            r (int): right row bound (inclusive, 0-based)
            c (int): right column bound (inclusive, 0-based)

        Returns:
            int: sum of all elements in region [0, r] x [0, c]

        """
        if r < 0 or c < 0:
            return 0

        result = 0
        i = r + 1  # Convert to 1-based index
        while i > 0:
            j = c + 1  # Convert to 1-based index
            while j > 0:
                result += self.tree[i][j]
                j -= j & -j
            i -= i & -i
        return result

    def get_array(self) -> list[list[int]]:
        """
        Get the actual array represented by the 2D Fenwick tree in O(R * C)

        Returns:
            list[list[int]]: 2D array of elements

        """
        arr: list[list[int]] = [[0] * self.cols for _ in range(self.rows)]
        for r in range(self.rows):
            for c in range(self.cols):
                arr[r][c] = self.query(r, c, r + 1, c + 1)
        return arr
