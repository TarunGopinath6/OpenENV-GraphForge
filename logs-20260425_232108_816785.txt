def scale(self, factor: int) -> None:
    """Scale all values in the Fenwick tree by ``factor`` in O(N).

    This multiplies every element of the underlying array by ``factor``.
    Since each Fenwick node stores a sum of some subset of elements,
    multiplying every stored partial sum by the same factor yields the
    correct scaled tree.

    Parameters:
        factor (int): multiplier to apply to every element

    Returns:
        None

    >>> f = FenwickTree([1, 2, 3, 4, 5])
    >>> f.scale(3)
    >>> f.get_array()
    [3, 6, 9, 12, 15]
    """
    # Each entry in `self.tree` is a partial sum; scaling all partial sums
    # scales the represented array elements as well.
    for i in range(self.size):
        self.tree[i] *= factor
