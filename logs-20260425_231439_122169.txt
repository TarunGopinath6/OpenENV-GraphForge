class FenwickTree2D:
    """
    2D Fenwick Tree (Fenwick Tree / Binary Indexed Tree)

    Supports point updates and rectangle range queries.
    Time complexity: O(log R * log C) for both operations.

    More info: https://en.wikipedia.org/wiki/Fenwick_tree#2D_Binary_Indexed_Tree
    """

    def __init__(self, rows: int, cols: int) -> None:
        """
        Constructor for the 2D Fenwick tree

        Parameters:
            rows (int): number of rows in the grid
            cols (int): number of columns in the grid

        Returns:
            None
        """
        self.rows = rows
        self.cols = cols
        # 1-indexed internally, so size is (rows+1) x (cols+1)
        self.tree = [[0] * (cols + 1) for _ in range(rows + 1)]

    def add(self, r: int, c: int, val: int) -> None:
        """
        Add a value to position (r, c) in O(log R * log C)

        The implementation is 0-indexed externally but converts to 1-indexed internally.

        Parameters:
            r (int): row index (0-indexed)
            c (int): column index (0-indexed)
            val (int): value to add

        Returns:
            None
        """
        # Convert to 1-indexed
        r += 1
        c += 1

        while r <= self.rows:
            new_r = r + (r & (-r))
            while new_r <= self.rows:
                new_c = c + (c & (-c))
                while new_c <= self.cols:
                    self.tree[new_r][new_c] += val
                    new_c = c
                r = new_r

    def query(self, r1: int, c1: int, r2: int, c2: int) -> int:
        """
        Query the sum of all elements in rectangle [r1, c1, r2, c2] inclusive

        Uses inclusion-exclusion principle with prefix sums.

        Parameters:
            r1 (int): top row (inclusive)
            c1 (int): left column (inclusive)
            r2 (int): bottom row (inclusive)
            c2 (int): right column (inclusive)

        Returns:
            int: sum of all elements in the rectangle
        """
        # Query prefix sum from (0,0) to (r, c) inclusive
        def prefix_sum(r: int, c: int) -> int:
            if r == 0 or c == 0:
                return 0
            sum_ = 0
            r += 1  # Convert to 1-indexed
            c += 1  # Convert to 1-indexed
            while r > 0:
                new_c = c
                while new_c > 0:
                    sum_ += self.tree[r][new_c]
                    new_c = new_c - (new_c & (-new_c))
                r = r - (r & (-r))
            return sum_

        # Inclusion-exclusion principle
        return (prefix_sum(r2, c2) - prefix_sum(r1 - 1, c2) -
                prefix_sum(r2, c1 - 1) + prefix_sum(r1 - 1, c1 - 1))
